# Grammar parse-smoke queries (openCypher 9 core)
MATCH (n:Person) RETURN n
MATCH (n:Person)-[:KNOWS]->(m) WHERE n.age > 30 RETURN m.name
MATCH (n:Person) WHERE (n)-[:KNOWS]->() RETURN n
RETURN [x IN range(1, 3) | x]
RETURN CASE WHEN 1 = 1 THEN 'a' ELSE 'b' END
CALL db.labels() YIELD label RETURN label
MATCH (n:Person) OPTIONAL MATCH (n)-[:KNOWS]->(m) RETURN n, m
UNWIND [1, 2] AS x RETURN x
MERGE (n:Person {name: 'Ada'}) RETURN n
CREATE (n:Person {name: 'Ada'}) RETURN n
MATCH (n) DELETE n
MATCH (n) SET n.x = 1 RETURN n
MATCH (n) REMOVE n.x RETURN n
